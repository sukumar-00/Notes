- name: Install Apache2
  hosts: nodes
  become: true

  tasks:
    - name: Update package cache
      ansible.builtin.apt:
        update_cache: true

    - name: Install Apache2
      ansible.builtin.apt:
        name: apache2
        state: present

    - name: Start and enable Apache2
      ansible.builtin.service:
        name: apache2
        state: started
        enabled: true
